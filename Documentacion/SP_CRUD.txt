SP LOGIN USUARIO
==============================================================*/

IF OBJECT_ID('sp_LoginUsuario', 'P') IS NOT NULL
    DROP PROCEDURE sp_LoginUsuario;
GO

CREATE PROCEDURE sp_LoginUsuario
(
    @Usuario VARCHAR(50)
)
AS
BEGIN

    SET NOCOUNT ON;

    SELECT
        U.Id,
        U.NombreCompleto,
        U.Usuario,
        U.Email,
        U.PasswordHash,
        R.Nombre AS Rol
    FROM Usuarios U
    INNER JOIN Roles R
        ON U.RolId = R.Id
    WHERE U.Usuario = @Usuario
    AND U.Activo = 1;

END
GO

/*==============================================================
SP PRODUCTOS LISTAR
==============================================================*/

IF OBJECT_ID('sp_Productos_Listar', 'P') IS NOT NULL
    DROP PROCEDURE sp_Productos_Listar;
GO

CREATE PROCEDURE sp_Productos_Listar
AS
BEGIN

    SET NOCOUNT ON;

    SELECT
        P.Id,
        P.Codigo,
        P.Nombre,
        P.Descripcion,

        C.Nombre AS Categoria,
        M.Nombre AS Marca,

        P.PrecioCompra,
        P.PrecioVenta,

        ISNULL(SUM(I.StockActual),0) AS Stock,

        P.StockMinimo,
        P.Activo

    FROM Productos P
    INNER JOIN Categorias C
        ON P.CategoriaId = C.Id
    INNER JOIN Marcas M
        ON P.MarcaId = M.Id
    LEFT JOIN Inventario I
        ON P.Id = I.ProductoId

    GROUP BY
        P.Id,
        P.Codigo,
        P.Nombre,
        P.Descripcion,
        C.Nombre,
        M.Nombre,
        P.PrecioCompra,
        P.PrecioVenta,
        P.StockMinimo,
        P.Activo

    ORDER BY P.Nombre;

END
GO

/*==============================================================
SP COMPRA REGISTRAR
==============================================================*/

IF OBJECT_ID('sp_Compra_Registrar', 'P') IS NOT NULL
    DROP PROCEDURE sp_Compra_Registrar;
GO

CREATE PROCEDURE sp_Compra_Registrar
(
    @ProveedorId INT,
    @UsuarioId INT,
    @TipoDocumento VARCHAR(20),
    @Serie VARCHAR(10),
    @NumeroDocumento VARCHAR(20),
    @SubTotal DECIMAL(18,2),
    @IGV DECIMAL(18,2),
    @Total DECIMAL(18,2),
    @ProductoId INT,
    @AlmacenId INT,
    @Cantidad INT,
    @PrecioUnitario DECIMAL(18,2)
)
AS
BEGIN

    SET NOCOUNT ON;

    BEGIN TRY

        BEGIN TRANSACTION;

        DECLARE @CompraId INT;
        DECLARE @StockAnterior INT;
        DECLARE @StockNuevo INT;

        INSERT INTO Compras
        (
            ProveedorId,
            UsuarioId,
            TipoDocumento,
            Serie,
            NumeroDocumento,
            SubTotal,
            IGV,
            Total
        )
        VALUES
        (
            @ProveedorId,
            @UsuarioId,
            @TipoDocumento,
            @Serie,
            @NumeroDocumento,
            @SubTotal,
            @IGV,
            @Total
        );

        SET @CompraId = SCOPE_IDENTITY();

        INSERT INTO CompraDetalle
        (
            CompraId,
            ProductoId,
            Cantidad,
            PrecioUnitario,
            Total
        )
        VALUES
        (
            @CompraId,
            @ProductoId,
            @Cantidad,
            @PrecioUnitario,
            @Cantidad * @PrecioUnitario
        );

        IF EXISTS
        (
            SELECT 1
            FROM Inventario
            WHERE ProductoId = @ProductoId
            AND AlmacenId = @AlmacenId
        )
        BEGIN

            SELECT @StockAnterior = StockActual
            FROM Inventario
            WHERE ProductoId = @ProductoId
            AND AlmacenId = @AlmacenId;

            UPDATE Inventario
            SET
                StockActual = StockActual + @Cantidad,
                UltimaActualizacion = GETDATE()
            WHERE ProductoId = @ProductoId
            AND AlmacenId = @AlmacenId;

        END
        ELSE
        BEGIN

            SET @StockAnterior = 0;

            INSERT INTO Inventario
            (
                ProductoId,
                AlmacenId,
                StockActual
            )
            VALUES
            (
                @ProductoId,
                @AlmacenId,
                @Cantidad
            );

        END

        SET @StockNuevo = @StockAnterior + @Cantidad;

        INSERT INTO Kardex
        (
            ProductoId,
            AlmacenId,
            TipoMovimiento,
            Referencia,
            ReferenciaId,
            Cantidad,
            StockAnterior,
            StockNuevo,
            CostoUnitario,
            UsuarioId
        )
        VALUES
        (
            @ProductoId,
            @AlmacenId,
            'COMPRA',
            'COMPRA',
            @CompraId,
            @Cantidad,
            @StockAnterior,
            @StockNuevo,
            @PrecioUnitario,
            @UsuarioId
        );

        COMMIT TRANSACTION;

        SELECT @CompraId AS CompraId;

    END TRY

    BEGIN CATCH

        ROLLBACK TRANSACTION;

        DECLARE @MensajeError NVARCHAR(4000);
        SET @MensajeError = ERROR_MESSAGE();

        RAISERROR(@MensajeError,16,1);

    END CATCH

END
GO

/*==============================================================
SP VENTA REGISTRAR
==============================================================*/

IF OBJECT_ID('sp_Venta_Registrar', 'P') IS NOT NULL
    DROP PROCEDURE sp_Venta_Registrar;
GO

CREATE PROCEDURE sp_Venta_Registrar
(
    @ClienteId INT,
    @UsuarioId INT,
    @TipoDocumento VARCHAR(20),
    @Serie VARCHAR(10),
    @NumeroDocumento VARCHAR(20),
    @SubTotal DECIMAL(18,2),
    @IGV DECIMAL(18,2),
    @Total DECIMAL(18,2),
    @ProductoId INT,
    @AlmacenId INT,
    @Cantidad INT,
    @PrecioUnitario DECIMAL(18,2)
)
AS
BEGIN

    SET NOCOUNT ON;

    BEGIN TRY

        BEGIN TRANSACTION;

        DECLARE @VentaId INT;
        DECLARE @StockActual INT;
        DECLARE @StockNuevo INT;

        SELECT @StockActual = StockActual
        FROM Inventario
        WHERE ProductoId = @ProductoId
        AND AlmacenId = @AlmacenId;

        IF @StockActual < @Cantidad
        BEGIN
            RAISERROR('Stock insuficiente.',16,1);
            RETURN;
        END

        INSERT INTO Ventas
        (
            ClienteId,
            UsuarioId,
            TipoDocumento,
            Serie,
            NumeroDocumento,
            SubTotal,
            IGV,
            Total
        )
        VALUES
        (
            @ClienteId,
            @UsuarioId,
            @TipoDocumento,
            @Serie,
            @NumeroDocumento,
            @SubTotal,
            @IGV,
            @Total
        );

        SET @VentaId = SCOPE_IDENTITY();

        INSERT INTO VentaDetalle
        (
            VentaId,
            ProductoId,
            Cantidad,
            PrecioUnitario,
            Total
        )
        VALUES
        (
            @VentaId,
            @ProductoId,
            @Cantidad,
            @PrecioUnitario,
            @Cantidad * @PrecioUnitario
        );

        UPDATE Inventario
        SET
            StockActual = StockActual - @Cantidad,
            UltimaActualizacion = GETDATE()
        WHERE ProductoId = @ProductoId
        AND AlmacenId = @AlmacenId;

        SET @StockNuevo = @StockActual - @Cantidad;

        INSERT INTO Kardex
        (
            ProductoId,
            AlmacenId,
            TipoMovimiento,
            Referencia,
            ReferenciaId,
            Cantidad,
            StockAnterior,
            StockNuevo,
            CostoUnitario,
            UsuarioId
        )
        VALUES
        (
            @ProductoId,
            @AlmacenId,
            'VENTA',
            'VENTA',
            @VentaId,
            @Cantidad,
            @StockActual,
            @StockNuevo,
            @PrecioUnitario,
            @UsuarioId
        );

        COMMIT TRANSACTION;

        SELECT @VentaId AS VentaId;

    END TRY

    BEGIN CATCH

        ROLLBACK TRANSACTION;

        DECLARE @MensajeError NVARCHAR(4000);
        SET @MensajeError = ERROR_MESSAGE();

        RAISERROR(@MensajeError,16,1);

    END CATCH

END
GO

/*==============================================================
SP PRODUCTOS STOCK BAJO
==============================================================*/

IF OBJECT_ID('sp_Productos_StockBajo', 'P') IS NOT NULL
    DROP PROCEDURE sp_Productos_StockBajo;
GO

CREATE PROCEDURE sp_Productos_StockBajo
AS
BEGIN

    SET NOCOUNT ON;

    SELECT
        P.Id,
        P.Codigo,
        P.Nombre,

        ISNULL(SUM(I.StockActual),0) AS Stock,

        P.StockMinimo

    FROM Productos P
    LEFT JOIN Inventario I
        ON P.Id = I.ProductoId

    GROUP BY
        P.Id,
        P.Codigo,
        P.Nombre,
        P.StockMinimo

    HAVING ISNULL(SUM(I.StockActual),0) <= P.StockMinimo

    ORDER BY Stock ASC;

END
GO

AS
BEGIN

    SET NOCOUNT ON;
