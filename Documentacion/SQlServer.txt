CREATE DATABASE ERP_REPUESTOS;
GO

USE ERP_REPUESTOS;
GO

/*==============================================================
 TABLA: ROLES
==============================================================*/

CREATE TABLE Roles(
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Nombre VARCHAR(50) NOT NULL UNIQUE,
    Descripcion VARCHAR(250),
    Activo BIT NOT NULL DEFAULT 1,
    FechaCreacion DATETIME NOT NULL DEFAULT GETDATE()
);
GO

/*==============================================================
 TABLA: USUARIOS
==============================================================*/

CREATE TABLE Usuarios(
    Id INT IDENTITY(1,1) PRIMARY KEY,
    RolId INT NOT NULL,

    NombreCompleto VARCHAR(150) NOT NULL,
    Usuario VARCHAR(50) NOT NULL UNIQUE,
    Email VARCHAR(150) NOT NULL UNIQUE,

    PasswordHash VARCHAR(500) NOT NULL,

    Telefono VARCHAR(20),

    IntentosFallidos INT DEFAULT 0,

    Activo BIT NOT NULL DEFAULT 1,

    FechaCreacion DATETIME NOT NULL DEFAULT GETDATE(),
    FechaActualizacion DATETIME NULL,

    FOREIGN KEY (RolId) REFERENCES Roles(Id)
);
GO

/*==============================================================
 TABLA: CLIENTES
==============================================================*/

CREATE TABLE Clientes(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    TipoDocumento VARCHAR(10) NOT NULL,
    NumeroDocumento VARCHAR(20) NOT NULL UNIQUE,

    Nombres VARCHAR(150) NOT NULL,

    Direccion VARCHAR(250),

    Telefono VARCHAR(20),
    Email VARCHAR(150),

    LimiteCredito DECIMAL(18,2) DEFAULT 0,
    Saldo DECIMAL(18,2) DEFAULT 0,

    Activo BIT NOT NULL DEFAULT 1,

    FechaRegistro DATETIME NOT NULL DEFAULT GETDATE()
);
GO

/*==============================================================
 TABLA: VEHICULOS
==============================================================*/

CREATE TABLE Vehiculos(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    ClienteId INT NOT NULL,

    Marca VARCHAR(100) NOT NULL,
    Modelo VARCHAR(100) NOT NULL,

    Anio INT,
    Color VARCHAR(50),

    Placa VARCHAR(20) UNIQUE,

    VIN VARCHAR(100),
    Motor VARCHAR(100),

    Kilometraje INT DEFAULT 0,

    FechaRegistro DATETIME DEFAULT GETDATE(),

    FOREIGN KEY (ClienteId) REFERENCES Clientes(Id)
);
GO

/*==============================================================
 TABLA: CATEGORIAS
==============================================================*/

CREATE TABLE Categorias(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    Nombre VARCHAR(100) NOT NULL UNIQUE,
    Descripcion VARCHAR(250),

    Activo BIT DEFAULT 1
);
GO

/*==============================================================
 TABLA: MARCAS
==============================================================*/

CREATE TABLE Marcas(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    Nombre VARCHAR(100) NOT NULL UNIQUE,

    Activo BIT DEFAULT 1
);
GO

/*==============================================================
 TABLA: PRODUCTOS
==============================================================*/

CREATE TABLE Productos(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    CategoriaId INT NOT NULL,
    MarcaId INT NOT NULL,

    Codigo VARCHAR(50) NOT NULL UNIQUE,
    CodigoOEM VARCHAR(100),

    Nombre VARCHAR(200) NOT NULL,
    Descripcion VARCHAR(500),

    UnidadMedida VARCHAR(20) DEFAULT 'UND',

    PrecioCompra DECIMAL(18,2) NOT NULL DEFAULT 0,
    PrecioVenta DECIMAL(18,2) NOT NULL DEFAULT 0,

    StockMinimo INT DEFAULT 0,
    StockMaximo INT DEFAULT 0,

    ManejaLotes BIT DEFAULT 0,
    ManejaSeries BIT DEFAULT 0,

    ImagenUrl VARCHAR(500),

    Peso DECIMAL(18,2),

    Activo BIT DEFAULT 1,

    FechaCreacion DATETIME DEFAULT GETDATE(),

    FOREIGN KEY (CategoriaId) REFERENCES Categorias(Id),
    FOREIGN KEY (MarcaId) REFERENCES Marcas(Id)
);
GO

/*==============================================================
 TABLA: ALMACENES
==============================================================*/

CREATE TABLE Almacenes(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    Nombre VARCHAR(100) NOT NULL,
    Direccion VARCHAR(250),

    Responsable VARCHAR(150),

    Activo BIT DEFAULT 1
);
GO

/*==============================================================
 TABLA: INVENTARIO
==============================================================*/

CREATE TABLE Inventario(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    ProductoId INT NOT NULL,
    AlmacenId INT NOT NULL,

    StockActual INT NOT NULL DEFAULT 0,

    CostoPromedio DECIMAL(18,2) DEFAULT 0,

    UltimaActualizacion DATETIME DEFAULT GETDATE(),

    FOREIGN KEY (ProductoId) REFERENCES Productos(Id),
    FOREIGN KEY (AlmacenId) REFERENCES Almacenes(Id),

    CONSTRAINT UQ_Inventario UNIQUE(ProductoId, AlmacenId)
);
GO

/*==============================================================
 TABLA: PROVEEDORES
==============================================================*/

CREATE TABLE Proveedores(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    TipoDocumento VARCHAR(10),
    NumeroDocumento VARCHAR(20),

    RazonSocial VARCHAR(200) NOT NULL,

    Direccion VARCHAR(250),

    Telefono VARCHAR(20),
    Email VARCHAR(150),

    Contacto VARCHAR(150),

    Activo BIT DEFAULT 1,

    FechaRegistro DATETIME DEFAULT GETDATE()
);
GO

/*==============================================================
 TABLA: COMPRAS
==============================================================*/

CREATE TABLE Compras(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    ProveedorId INT NOT NULL,
    UsuarioId INT NOT NULL,

    Fecha DATETIME NOT NULL DEFAULT GETDATE(),

    TipoDocumento VARCHAR(20),
    Serie VARCHAR(10),
    NumeroDocumento VARCHAR(20),

    SubTotal DECIMAL(18,2) NOT NULL,
    IGV DECIMAL(18,2) NOT NULL,
    Total DECIMAL(18,2) NOT NULL,

    Observacion VARCHAR(500),

    Estado VARCHAR(20) DEFAULT 'REGISTRADO',

    FOREIGN KEY (ProveedorId) REFERENCES Proveedores(Id),
    FOREIGN KEY (UsuarioId) REFERENCES Usuarios(Id)
);
GO

/*==============================================================
 TABLA: COMPRA DETALLE
==============================================================*/

CREATE TABLE CompraDetalle(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    CompraId INT NOT NULL,
    ProductoId INT NOT NULL,

    Cantidad INT NOT NULL,

    PrecioUnitario DECIMAL(18,2) NOT NULL,

    Descuento DECIMAL(18,2) DEFAULT 0,

    Total DECIMAL(18,2) NOT NULL,

    FOREIGN KEY (CompraId) REFERENCES Compras(Id),
    FOREIGN KEY (ProductoId) REFERENCES Productos(Id)
);
GO

/*==============================================================
 TABLA: VENTAS
==============================================================*/

CREATE TABLE Ventas(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    ClienteId INT NOT NULL,
    UsuarioId INT NOT NULL,

    Fecha DATETIME DEFAULT GETDATE(),

    TipoDocumento VARCHAR(20),
    Serie VARCHAR(10),
    NumeroDocumento VARCHAR(20),

    SubTotal DECIMAL(18,2),
    IGV DECIMAL(18,2),
    Total DECIMAL(18,2),

    MetodoPago VARCHAR(50),

    Estado VARCHAR(20) DEFAULT 'REGISTRADO',

    Observacion VARCHAR(500),

    FOREIGN KEY (ClienteId) REFERENCES Clientes(Id),
    FOREIGN KEY (UsuarioId) REFERENCES Usuarios(Id)
);
GO

/*==============================================================
 TABLA: VENTA DETALLE
==============================================================*/

CREATE TABLE VentaDetalle(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    VentaId INT NOT NULL,
    ProductoId INT NOT NULL,

    Cantidad INT NOT NULL,

    PrecioUnitario DECIMAL(18,2) NOT NULL,

    Descuento DECIMAL(18,2) DEFAULT 0,

    Total DECIMAL(18,2) NOT NULL,

    FOREIGN KEY (VentaId) REFERENCES Ventas(Id),
    FOREIGN KEY (ProductoId) REFERENCES Productos(Id)
);
GO

/*==============================================================
 TABLA: KARDEX
==============================================================*/

CREATE TABLE Kardex(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    ProductoId INT NOT NULL,
    AlmacenId INT NOT NULL,

    TipoMovimiento VARCHAR(20) NOT NULL,

    Referencia VARCHAR(50),
    ReferenciaId INT,

    Cantidad INT NOT NULL,

    StockAnterior INT NOT NULL,
    StockNuevo INT NOT NULL,

    CostoUnitario DECIMAL(18,2),

    Fecha DATETIME DEFAULT GETDATE(),

    UsuarioId INT,

    FOREIGN KEY (ProductoId) REFERENCES Productos(Id),
    FOREIGN KEY (AlmacenId) REFERENCES Almacenes(Id),
    FOREIGN KEY (UsuarioId) REFERENCES Usuarios(Id)
);
GO

/*==============================================================
 TABLA: ALERTAS CLIENTES
==============================================================*/

CREATE TABLE AlertasClientes(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    ClienteId INT NOT NULL,
    VehiculoId INT NULL,

    TipoAlerta VARCHAR(50),

    Titulo VARCHAR(150),

    Mensaje VARCHAR(MAX),

    FechaProgramada DATETIME,

    Enviado BIT DEFAULT 0,

    FechaEnvio DATETIME NULL,

    MedioEnvio VARCHAR(20),

    FOREIGN KEY (ClienteId) REFERENCES Clientes(Id),
    FOREIGN KEY (VehiculoId) REFERENCES Vehiculos(Id)
);
GO

/*==============================================================
 TABLA: AUDITORIA
==============================================================*/

CREATE TABLE Auditoria(
    Id BIGINT IDENTITY(1,1) PRIMARY KEY,

    UsuarioId INT,

    Tabla VARCHAR(100),
    Accion VARCHAR(50),

    RegistroId INT,

    DatosAnteriores VARCHAR(MAX),
    DatosNuevos VARCHAR(MAX),

    Fecha DATETIME DEFAULT GETDATE(),

    IP VARCHAR(50),

    FOREIGN KEY (UsuarioId) REFERENCES Usuarios(Id)
);
GO

/*==============================================================
 TABLA: CONFIGURACION EMPRESA
==============================================================*/

CREATE TABLE Empresa(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    RazonSocial VARCHAR(200),
    NombreComercial VARCHAR(200),

    RUC VARCHAR(20),

    Direccion VARCHAR(250),

    Telefono VARCHAR(20),
    Email VARCHAR(150),

    LogoUrl VARCHAR(500),

    IGV DECIMAL(18,2) DEFAULT 18
);
GO

/*==============================================================
 TABLA: CAJA
==============================================================*/

CREATE TABLE Caja(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    UsuarioId INT NOT NULL,

    FechaApertura DATETIME DEFAULT GETDATE(),
    FechaCierre DATETIME NULL,

    MontoInicial DECIMAL(18,2),
    MontoFinal DECIMAL(18,2),

    Estado VARCHAR(20) DEFAULT 'ABIERTA',

    FOREIGN KEY (UsuarioId) REFERENCES Usuarios(Id)
);
GO

/*==============================================================
 TABLA: MOVIMIENTO CAJA
==============================================================*/

CREATE TABLE MovimientoCaja(
    Id INT IDENTITY(1,1) PRIMARY KEY,

    CajaId INT NOT NULL,

    TipoMovimiento VARCHAR(20),

    Concepto VARCHAR(200),

    Monto DECIMAL(18,2),

    Fecha DATETIME DEFAULT GETDATE(),

    FOREIGN KEY (CajaId) REFERENCES Caja(Id)
);
GO

/*==============================================================
 INDICES
==============================================================*/

CREATE INDEX IX_Productos_Codigo
ON Productos(Codigo);

CREATE INDEX IX_Productos_Nombre
ON Productos(Nombre);

CREATE INDEX IX_Kardex_Producto
ON Kardex(ProductoId);

CREATE INDEX IX_Kardex_Fecha
ON Kardex(Fecha);

CREATE INDEX IX_Ventas_Fecha
ON Ventas(Fecha);

CREATE INDEX IX_Compras_Fecha
ON Compras(Fecha);

CREATE INDEX IX_Clientes_Documento
ON Clientes(NumeroDocumento);

GO
